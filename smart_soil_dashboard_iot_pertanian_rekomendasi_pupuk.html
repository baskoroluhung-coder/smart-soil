SmartSoil_Node01.ino 
#include <WiFi.h>
#include <HTTPClient.h>
#include <OneWire.h>
#include <DallasTemperature.h>

// Konfigurasi WiFi & Endpoint Smart Soil
const char* ssid = "NAMA_WIFI_ANDA";
const char* password = "PASSWORD_WIFI_ANDA";
const char* serverUrl = "https://api.smartsoil.id/v1/telemetry";
const char* deviceId = "SmartSoil-Node-01";

// Definisi Pin Sensor ESP32
#define PIN_MOISTURE 34
#define PIN_PH       35
#define PIN_DS18B20  4

OneWire oneWire(PIN_DS18B20);
DallasTemperature tempSensor(&oneWire);

void setup() {
  Serial.begin(115200);
  tempSensor.begin();
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\n[Smart Soil] Terhubung ke WiFi!");
}

void loop() {
  // 1. Baca Kelembapan Tanah (Kalibrasi nilai kering=3200, basah=1400)
  int rawMoisture = analogRead(PIN_MOISTURE);
  float moisturePct = constrain(map(rawMoisture, 3200, 1400, 0, 100), 0, 100);

  // 2. Baca Suhu Tanah DS18B20 (°C)
  tempSensor.requestTemperatures();
  float soilTemp = tempSensor.getTempCByIndex(0);

  // 3. Baca pH Tanah (Konversi tegangan 0-3.3V ke skala pH 3.5-9.0)
  float voltagePh = analogRead(PIN_PH) * (3.3 / 4095.0);
  float soilPh = constrain(3.5 + (voltagePh * 1.65), 3.5, 9.0);

  // 4. Kirim Payload JSON ke Dashboard Smart Soil
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(serverUrl);
    http.addHeader("Content-Type", "application/json");

    String payload = "{\"device_id\":\"" + String(deviceId) + "\","
                   + "\"moisture\":" + String(moisturePct, 1) + ","
                   + "\"temperature\":" + String(soilTemp, 1) + ","
                   + "\"ph\":" + String(soilPh, 2) + "}";

    int httpCode = http.POST(payload);
    Serial.printf("[Smart Soil] POST %d -> %s\n", httpCode, payload.c_str());
    http.end();
  }
  delay(5000); // Kirim telemetri setiap 5 detik
}
